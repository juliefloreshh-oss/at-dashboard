// Linear Regression Helper

function linearRegression(yVals) {

  const n = yVals.length;

  let sumX = 0, sumY = 0, sumXY = 0, sumXX = 0;

  for (let i = 0; i < n; i++) {

    sumX += i;

    sumY += yVals[i];

    sumXY += i * yVals[i];

    sumXX += i * i;

  }

  const slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX || 1);

  const intercept = (sumY - slope * sumX) / (n || 1);

  return { slope, intercept };

}

        backgroundColor: barColors,

        borderRadius: 6

      }]

    },

    options: {

      responsive: true,

      maintainAspectRatio: false,

      plugins: { legend: { display: false } },

      scales: {

        x: { ticks: { color: '#8b949e' }, grid: { color: '#21262d' } },

        y: { ticks: { color: '#8b949e' }, grid: { color: '#21262d' }, beginAtZero: true }

      }

    }

  });


  // Location Donut Chart

  const locCounts = {};

  filtered.forEach(r => {

    const loc = r[4] || 'Unspecified';

    locCounts[loc] = (locCounts[loc] || 0) + 1;

  });

  const sortedLocs = Object.entries(locCounts).sort((a, b) => b[1] - a[1]);

  const topLocs = sortedLocs.slice(0, 6);

  const otherSum = sortedLocs.slice(6).reduce((acc, cur) => acc + cur[1], 0);

  if (otherSum > 0) topLocs.push(['Other Types', otherSum]);


  if (locationDonutChart) locationDonutChart.destroy();

  locationDonutChart = new Chart(document.getElementById('locationDonutChart'), {

    data: {

      datasets: [{

        borderColor: '#161b22',

        borderWidth: 2

      }]

    },

    options: {

      responsive: true,

      plugins: {

        legend: { position: 'right', labels: { color: '#f0f6fc', boxWidth: 12 } }

      }

    }

  });

}


function updateTrends(filtered) {

  const years = [2010, 2011, 2012, 2013, 2014, 2015, 2016, 2017, 2018];

  const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];

  const fullMonths = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];


  // Build month-by-month series (2010-01 to 2018-12)

  const ymMap = {};

  filtered.forEach(r => {

    if (r[1]) ymMap[r[1]] = (ymMap[r[1]] || 0) + 1;

  });


  const labels = [];

  const actuals = [];

  years.forEach(y => {

    for (let m = 1; m <= 12; m++) {

      const ym = `${y}-${String(m).padStart(2, '0')}`;

      labels.push(ym);

      actuals.push(ymMap[ym] || 0);

    }

  });


  // Trim trailing zero months in 2018 if dataset ends mid-year

  while (actuals.length > 12 && actuals[actuals.length - 1] === 0) {

    actuals.pop();

    labels.pop();

  }


  const { slope, intercept } = linearRegression(actuals);

  const direction = slope >= 0 ? 'Upward ▲' : 'Downward ▼';

  document.getElementById('trend-badge').textContent =

    `Trend: ${direction} (${slope >= 0 ? '+' : ''}${slope.toFixed(2)} sightings/month)`;


  // Extend 12 months into forecast

  const forecastHorizon = 12;

  const extendedLabels = [...labels];

  const actualSeries = [...actuals];

  const trendSeries = actuals.map((_, i) => Math.max(0, intercept + slope * i));


  let [lastY, lastM] = labels[labels.length - 1].split('-').map(Number);

  for (let f = 1; f <= forecastHorizon; f++) {

    lastM++;

    if (lastM > 12) { lastM = 1; lastY++; }

    extendedLabels.push(`${lastY}-${String(lastM).padStart(2, '0')} (F)`);

    actualSeries.push(null);

    trendSeries.push(Math.max(0, intercept + slope * (actuals.length - 1 + f)));

  }


  if (monthlyLineChart) monthlyLineChart.destroy();

  monthlyLineChart = new Chart(document.getElementById('monthlyLineChart'), {

    type: 'line',

    data: {

      labels: extendedLabels,

      datasets: [

        {

          label: 'Monthly Sightings',

          data: actualSeries,

          borderColor: '#58a6ff',

          backgroundColor: 'rgba(88, 166, 255, 0.15)',

          fill: true,

          tension: 0.25,

          pointRadius: 2

        },

        {

          label: 'Linear Regression & 12-Mo Forecast',

          data: trendSeries,

          borderColor: '#f85149',

          borderDash: [6, 6],

          borderWidth: 2,

          pointRadius: 0,

          fill: false

        }

      ]

    },

    options: {

      responsive: true,

      maintainAspectRatio: false,

      plugins: { legend: { labels: { color: '#f0f6fc' } } },

      scales: {

        x: { ticks: { color: '#8b949e', maxTicksLimit: 18 }, grid: { color: '#21262d' } },

        y: { ticks: { color: '#8b949e' }, grid: { color: '#21262d' }, beginAtZero: true }

      }

    }

  });


  // Seasonal Heatmap Table

  let maxCell = 1;

  const grid = Array.from({ length: 12 }, () => Array(years.length).fill(0));

  filtered.forEach(r => {

    if (!r[1]) return;

    const [yStr, mStr] = r[1].split('-');

    const yIdx = years.indexOf(Number(yStr));

    const mIdx = Number(mStr) - 1;

    if (yIdx >= 0 && mIdx >= 0 && mIdx < 12) {

      grid[mIdx][yIdx]++;

      if (grid[mIdx][yIdx] > maxCell) maxCell = grid[mIdx][yIdx];

    }

  });


  let html = '<table class="heatmap"><thead><tr><th>Month</th>';

  years.forEach(y => { html += `<th>${y}</th>`; });

  html += '</tr></thead><tbody>';


  for (let m = 0; m < 12; m++) {

    html += `<tr><td class="month-col">${fullMonths[m]}</td>`;

    for (let y = 0; y < years.length; y++) {

      const val = grid[m][y];

      const intensity = val / maxCell;

      const bg = val === 0

        ? 'rgba(33, 38, 45, 0.4)'

        : `rgba(88, 166, 255, ${(0.15 + intensity * 0.8).toFixed(2)})`;

      const textColor = intensity > 0.55 ? '#0d1117' : '#f0f6fc';

      html += `<td style="background:${bg}; color:${textColor}; font-weight:${intensity > 0.55 ? '700' : '400'};">${val}</td>`;

    }

    html += '</tr>';

  }

  html += '</tbody></table>';

  document.getElementById('heatmapContainer').innerHTML = html;

}


function initMap() {

  map = L.map('nyc-map').setView([40.7128, -73.9560], 11);

  L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {

    attribution: '&copy; OpenStreetMap &copy; CARTO',

    maxZoom: 18

  }).addTo(map);


  zipLayer = L.layerGroup().addTo(map);

  pinLayer = L.layerGroup();


  legendControl = L.control({ position: 'bottomright' });

  legendControl.onAdd = function () {

    const div = L.DomUtil.create('div', 'legend');

    div.id = 'map-legend';

    return div;

  };

  legendControl.addTo(map);


  document.getElementById('btnZipView').addEventListener('click', () => {

    showSamplePins = false;

    document.getElementById('btnZipView').classList.add('active');

    document.getElementById('btnPinView').classList.remove('active');

    updateMap(getFilteredData());

  });


  document.getElementById('btnPinView').addEventListener('click', () => {

    showSamplePins = !showSamplePins;

    document.getElementById('btnPinView').classList.toggle('active', showSamplePins);

    updateMap(getFilteredData());

  });

}


function getZipColor(count, maxCount) {

  const ratio = count / (maxCount || 1);

  if (ratio > 0.75) return '#08519c';

  if (ratio > 0.50) return '#3182bd';

  if (ratio > 0.25) return '#6baed6';

  return '#bdd7e7';

}


function updateMap(filtered) {

  if (!map) initMap();

  zipLayer.clearLayers();

  pinLayer.clearLayers();


  // Aggregate by Zip Code

  const zipAgg = {};

  filtered.forEach(r => {

    const zip = r[5], lat = r[6], lng = r[7];

    if (!zip || !lat || !lng) return;

    if (!zipAgg[zip]) zipAgg[zip] = { count: 0, latSum: 0, lngSum: 0, borough: r[0] };

    zipAgg[zip].count++;

    zipAgg[zip].latSum += lat;

    zipAgg[zip].lngSum += lng;

  });


  const zipEntries = Object.entries(zipAgg);

  const maxZip = Math.max(1, ...zipEntries.map(z => z[1].count));


  zipEntries.forEach(([zip, info]) => {

    const cLat = info.latSum / info.count;

    const cLng = info.lngSum / info.count;

    const color = getZipColor(info.count, maxZip);

    const radius = 6 + (info.count / maxZip) * 16;


    L.circleMarker([cLat, cLng], {

      radius,

      fillColor: color,

      color: '#f0f6fc',

      weight: 1,

      opacity: 0.85,

      fillOpacity: 0.8

    })

    .bindPopup(`<strong>ZIP: ${zip}</strong> (${info.borough})<br/>Total Sightings: <b>${info.count}</b>`)

    .addTo(zipLayer);

  });


  // Update Legend

  const q1 = Math.max(1, Math.round(maxZip * 0.25));

  const q2 = Math.max(2, Math.round(maxZip * 0.50));

  const q3 = Math.max(3, Math.round(maxZip * 0.75));

  document.getElementById('map-legend').innerHTML = `

    <strong>Sightings per Zip Code</strong>

    <div class="legend-item"><span class="legend-swatch" style="background:#bdd7e7"></span> 1 – ${q1}</div>

    <div class="legend-item"><span class="legend-swatch" style="background:#6baed6"></span> ${q1 + 1} – ${q2}</div>

    <div class="legend-item"><span class="legend-swatch" style="background:#3182bd"></span> ${q2 + 1} – ${q3}</div>

    <div class="legend-item"><span class="legend-swatch" style="background:#08519c"></span> ${q3 + 1}+ (Peak)</div>

  `;


  // Sample Pins Layer

  if (showSamplePins) {

    const sample = filtered.filter(r => r[6] && r[7]).slice(0, 300);

    sample.forEach(r => {

      L.marker([r[6], r[7]])

        .bindPopup(`

          <strong>${r[8] || 'Intersection / Blockface'}</strong><br/>

          Borough: ${r[0]} (${r[5] || 'N/A'})<br/>

          Date: ${r[1]} | Status: <b>${r[2]}</b><br/>

          Location: ${r[4] || 'Unspecified'}

        `)

        .addTo(pinLayer);

    });

    pinLayer.addTo(map);

  } else {

    map.removeLayer(pinLayer);

  }

}


function updateDashboard() {

  const filtered = getFilteredData();

  updateSummary(filtered);

  updateTrends(filtered);

  updateMap(filtered);

}


updateDashboard();
